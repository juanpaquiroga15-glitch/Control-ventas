async function registrarVenta() { let pId = document.getElementById('vProducto').value; let p = productos[pId]; let cant = parseInt(document.getElementById('vCantidad').value) || 1; // Convertir todos los valores a números explícitos para evitar errores de cálculo let precioVenta = parseFloat(p.precio) || 0; let costoInsumo = parseFloat(p.costo) || 0; let valComision = parseFloat(p.valorComision) || 0; let totalVenta = precioVenta * cant; let totalCosto = costoInsumo * cant; let totalComision = 0; // Cálculo exacto de la comisión if (p.tipoComision === 'porcentaje') { // Si puso 25 (como 25%) o 0.25, el código ajusta el porcentaje automáticamente let porcentaje = valComision > 1 ? valComision / 100 : valComision; totalComision = totalVenta * porcentaje; } else { // Monto fijo por unidad vendida totalComision = valComision * cant; } // Ganancia Neta = Lo que ingresa - Costo de compra - Comisión del vendedor let gananciaNeta = totalVenta - totalCosto - totalComision; let reg = { fecha: new Date().toLocaleDateString() + ' ' + new Date().toLocaleTimeString([], {hour: '2-digit', minute:'2-digit'}), timestamp: Date.now(), usr: usuario, uni: document.getElementById('vUni').value, vendedor: document.getElementById('vVendedor').value.trim() || 'N/A', prod: p.nombre, cant: cant, total: totalVenta, com: totalComision, cos: totalCosto, gan: gananciaNeta }; if(db) { await db.collection("ventas").add(reg); } else { reg.id = 'local_' + Date.now(); ventas.unshift(reg); localStorage.setItem('vnts', JSON.stringify(ventas)); actualizarTabla(); } alert('Venta registrada con éxito'); }
